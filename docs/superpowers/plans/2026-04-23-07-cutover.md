# Plan 07: legacy URL redirects, cutover, and cleanup

> **For agentic workers:** REQUIRED SUB-SKILL: use `superpowers:subagent-driven-development` (recommended).
>
> **BEFORE STARTING, READ `docs/superpowers/plans/ERRATA.md` IN FULL.** Entries override this plan.
>

**Goal:** Preserve every old notebook URL by redirecting to an equivalent workspace URL, then delete the entire Python pipeline, notebooks, Astro remnants, and legacy assets. Rewrite the R2 upload script in TypeScript. CI runs Bun-only.

**Architecture:** A shared `legacy.ts` module (living under `worker/src/`, imported by both the site and the worker) maps old URL shapes to encoded workspace states. The site exposes `/latest/*` and `/:year/:month/:day[/:notebook_id]` routes that compute the redirect client-side (returning a `<Navigate replace />`); the worker handles them server-side for direct-hit traffic. Afterwards the Python tree, Astro artifacts, and retired queries are deleted in one cutover commit.

**Tech Stack:** React Router `<Navigate>`, the same `url.ts` encoder from Plan 06, the Cloudflare Worker runtime. `upload_r2.ts` uses the S3-compatible R2 API via `@aws-sdk/client-s3`.

```ts
// Shared by the Cloudflare Worker AND the Vite-built site.
// Keep this file dependency-free so it can be imported by either.

import pako from 'pako';

export const NOTEBOOK_PRESETS: Readonly<Record<string, readonly string[]>> = {
  'blob-inclusion': [
    'blob-density-scatter', 'blob-count-stacked-epoch',
    'blob-popularity-heatmap', 'blob-slot-heatmap-vertical',
  ],
  'blob-flow': [
    'entity-blobcount-sankey', 'relay-blobcount-sankey',
    'entity-relay-sankey', 'entity-relay-blobcount-sankey',
  ],
  'column-propagation': [
    'column-first-seen-heatmap', 'column-delta-heatmap',
    'column-spread-timeseries',
  ],
  'mempool-visibility': [
    'hourly-coverage-lines', 'coverage-heatmap',
    'age-percentile-lines', 'sentry-coverage-bar',
  ],
  'mev-pipeline': [
    'bid-vs-block-scatter', 'bid-value-vs-block', 'bidding-duration-vs-block',
  ],
  'block-column-timing': [
    'block-to-column-histogram', 'block-to-column-boxplot',
    'block-to-column-timeseries', 'column-spread-boxplot-blob',
  ],
  'propagation-anomalies': [
    'anomaly-regression-scatter', 'anomalies-by-relay-bar',
    'anomalies-by-blobcount-bar', 'anomalies-table',
  ],
  'missed-slots': [
    'missed-slots-by-entity-bar', 'entity-miss-rate-bar',
    'missed-slots-hourly-bar', 'missed-slots-timeline-scatter',
  ],
  'block-propagation-size': [
    'block-propagation-by-size', 'block-propagation-by-region',
    'corrected-vs-size-scatter', 'regional-cdf-subplots',
  ],
};

type Pane  = { kind: 'pane';  id: string; chartId: string; date: string };
type Split = { kind: 'split'; id: string; orientation: 'h' | 'v'; ratio: number; a: Node; b: Node };
type Node  = Pane | Split;

function pane(chartId: string, date: string, idx: number): Pane {
  return { kind: 'pane', id: `p${idx}`, chartId, date };
}

function horiz(a: Node, b: Node, idx: number, ratio = 0.5): Split {
  return { kind: 'split', id: `s${idx}`, orientation: 'h', ratio, a, b };
}

function vert(a: Node, b: Node, idx: number, ratio = 0.5): Split {
  return { kind: 'split', id: `s${idx}`, orientation: 'v', ratio, a, b };
}

function layoutFor(chartIds: readonly string[], date: string): Node | null {
  const panes = chartIds.map((c, i) => pane(c, date, i));
  if (panes.length === 0) return null;
  if (panes.length === 1) return panes[0]!;
  if (panes.length === 2) return horiz(panes[0]!, panes[1]!, 10);
  if (panes.length === 3) return horiz(panes[0]!, vert(panes[1]!, panes[2]!, 11), 10);
  // 4 or more: 2x2 grid of the first four
  return horiz(
    vert(panes[0]!, panes[1]!, 12),
    vert(panes[2]!, panes[3]!, 13),
    10,
  );
}

// Copy of site/src/workspace/state/url.ts encoder, duplicated here so the
// worker has no site-side dependency. If the encoder ever changes, change
// both in lockstep (add an ERRATA entry noting the sync).
type EncodedPane = ['p', number, number];
type EncodedSplit = ['s' | 'S', number, EncodedTree, EncodedTree];
type EncodedTree = EncodedPane | EncodedSplit;

function encodeNode(n: Node, charts: Map<string, number>, dates: Map<string, number>): EncodedTree {
  if (n.kind === 'pane') {
    const c = charts.get(n.chartId) ?? (charts.set(n.chartId, charts.size), charts.get(n.chartId))!;
    const d = dates.get(n.date)   ?? (dates.set(n.date, dates.size),     dates.get(n.date))!;
    return ['p', c, d];
  }
  return [
    n.orientation === 'h' ? 's' : 'S',
    Math.round(n.ratio * 100),
    encodeNode(n.a, charts, dates),
    encodeNode(n.b, charts, dates),
  ];
}

function toBase64Url(bytes: Uint8Array): string {
  let s = '';
  for (let i = 0; i < bytes.length; i++) s += String.fromCharCode(bytes[i]!);
  return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}

export interface LegacyTarget {
  notebookId?: string;
  date?: string;           // YYYY-MM-DD; undefined means "latest"
}

export function resolveLegacy(t: LegacyTarget, latestDate: string): string {
  const date = t.date ?? latestDate;
  const chartIds = t.notebookId ? NOTEBOOK_PRESETS[t.notebookId] : undefined;
  const tree = chartIds ? layoutFor(chartIds, date) : null;
  const charts = new Map<string, number>();
  const dates  = new Map<string, number>();
  const treeEnc = tree ? encodeNode(tree, charts, dates) : null;
  const defIdx = dates.get(date) ?? (dates.set(date, dates.size), dates.get(date)!);
  const payload = [[...charts.keys()], [...dates.keys()], defIdx, null, treeEnc];
  const bytes = pako.gzip(JSON.stringify(payload));
  return `/w/${toBase64Url(bytes)}`;
}
```

Note: the encoded shape above must match the schema in `site/src/workspace/state/url.ts`. If Plan 06's url.ts was modified during implementation (field order, etc.), reconcile here before running tests. Append to ERRATA if any field diverges.

```tsx
import { Navigate, useParams } from 'react-router';
import { useEffect, useState } from 'react';
import { resolveLegacy } from '../../../worker/src/legacy';
import { loadDates } from '@/workspace/data/dates';

export default function Legacy() {
  const params = useParams<{ year?: string; month?: string; day?: string; id?: string }>();
  const [latest, setLatest] = useState<string | null>(null);

  useEffect(() => { loadDates().then((d) => setLatest(d.latest)); }, []);

  if (!latest) return <div className="p-8 text-xs text-muted">resolving...</div>;

  const date = params.year && params.month && params.day
    ? `${params.year}-${params.month}-${params.day}`
    : undefined;
  const target = resolveLegacy({ date, notebookId: params.id }, latest);
  return <Navigate to={target} replace />;
}
```

```ts
// observatory/src/upload.ts
#!/usr/bin/env bun
import fs from 'node:fs/promises';
import path from 'node:path';
import crypto from 'node:crypto';
import { S3Client, PutObjectCommand, ListObjectsV2Command } from '@aws-sdk/client-s3';
import { Upload } from '@aws-sdk/lib-storage';
import mime from 'mime';
import fg from 'fast-glob';

interface Env {
  accountId: string;
  accessKeyId: string;
  secretAccessKey: string;
  bucket: string;
  manifestKey: string;   // e.g. manifests/main.json or manifests/pr-14.json
}

function readEnv(): Env {
  const e = process.env;
  for (const k of ['R2_ACCOUNT_ID', 'R2_ACCESS_KEY', 'R2_SECRET_KEY', 'R2_BUCKET', 'R2_MANIFEST_KEY']) {
    if (!e[k]) throw new Error(`${k} must be set`);
  }
  return {
    accountId: e.R2_ACCOUNT_ID!,
    accessKeyId: e.R2_ACCESS_KEY!,
    secretAccessKey: e.R2_SECRET_KEY!,
    bucket: e.R2_BUCKET!,
    manifestKey: e.R2_MANIFEST_KEY!,
  };
}

async function sha256File(p: string): Promise<string> {
  const h = crypto.createHash('sha256');
  h.update(await fs.readFile(p));
  return h.digest('hex');
}

async function main(): Promise<void> {
  const env = readEnv();
  const s3 = new S3Client({
    region: 'auto',
    endpoint: `https://${env.accountId}.r2.cloudflarestorage.com`,
    credentials: { accessKeyId: env.accessKeyId, secretAccessKey: env.secretAccessKey },
  });

  const distDir = 'site/dist';
  const files = await fg(['**/*'], { cwd: distDir, dot: false, onlyFiles: true });
  const manifest: Record<string, string> = {};

  for (const rel of files) {
    const abs = path.join(distDir, rel);
    const hash = await sha256File(abs);
    const ext = path.extname(rel);
    const blobKey = `blobs/${hash}${ext}`;
    manifest[`/${rel.replace(/index\.html$/, '')}`] = blobKey;

    // Put only if missing (idempotent)
    try {
      const existing = await s3.send(new ListObjectsV2Command({ Bucket: env.bucket, Prefix: blobKey, MaxKeys: 1 }));
      if (existing.Contents && existing.Contents.length > 0) continue;
    } catch { /* continue */ }

    const body = await fs.readFile(abs);
    const contentType = mime.getType(rel) ?? 'application/octet-stream';
    await new Upload({
      client: s3,
      params: {
        Bucket: env.bucket,
        Key: blobKey,
        Body: body,
        ContentType: contentType,
        CacheControl: 'public, max-age=31536000, immutable',
      },
    }).done();
    process.stdout.write(`+ ${blobKey}\n`);
  }

  // Put manifest
  await s3.send(new PutObjectCommand({
    Bucket: env.bucket,
    Key: env.manifestKey,
    Body: JSON.stringify(manifest, null, 2),
    ContentType: 'application/json',
    CacheControl: 'public, max-age=0, must-revalidate',
  }));
  process.stdout.write(`manifest: ${env.manifestKey}\n`);
}

main().catch((e) => { console.error(e); process.exit(1); });
```
